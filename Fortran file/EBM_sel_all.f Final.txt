      double precision tbegin,tend,step_prt,step_min,step_max,eps
      integer nyprt
      
      character*200 dirname
      
      write(*,*)'Enter the path of the directory to store the results'
      write(*,*)'(the path must end by "/")'      
      read(*,'(a200)')dirname
      
      call make_directory(dirname)

      write(*,*)'Enter total number of years for the integration' 
      read(*,*)nyear         

c    kdebug = flag for debugging 
c       kdebug = 1 printing every timestep
c       kdebug = 0 printing only after pre-defined time step (step_prt)       
      kdebug = 0

c    Full result file (last nyprt years)
      open(21,file=trim(dirname)//'EBM_results_earth_seasonal.res')

c    Result files (seasonal and yearly)
      open(10,file=trim(dirname)//'EBM_results_seasonal.res') 
      open(11,file=trim(dirname)//'EBM_results_yearly_mean.res')      

      write(10,110)'Time (yr)','iyr','m','dn','pCO2','xlum'

     &              ,'MTem -65','MTem -55','MTem -45','MTem -35'

c final time for integration (yr) 
      tend = dble(nyear)
c time step for printing (yr)     
      step_prt = 1./12.
c minimum step allowed in ODE solver      
      step_min = 1.0d-13   !0.0000001 !Martian condition     		Updated!
c maximum step allowed in ODE solver      
      step_max = step_prt
c requested precision in ODE solver     
      eps = 1.d-3        !eps = 1.d-9					Updated !
c number of years to print in unit 21      
      nyprt = 10
      
      time0 = tbegin-0.5*step_prt
      

      call ode(tbegin,tend,y,step_prt,step_min,step_max,eps,nyprt)
 
      call givedrv(tend,y,drv)  

      close(21)
      close(10)
      close(11)

      stop
      end

      return
      end

     1            ,FIR_m(NB,0:nl,0:nm),DFMER_m(NB,0:nl,0:nm)

     1            ,TEMGL_m(0:nm),ALBGL_m(0:nm),ALBSGL_m(0:nm)
     1            ,nyr,nyrprv,nmth,mlength(nm),imonth(nday)
     1            ,nummth(0:nm)
      double precision temp_m,PTEM_m,ALB_m,SOL_m,FIR_m,DFMER_m,fnet_m
     1            ,FMER_m,FMSEN_m,FMLAT_m,FMOC_m,fi_m,fsnw_m,TEMGL_m
     2            ,ALBGL_m,ALBSGL_m
      integer nyr,nyrprv,nmth,mlength,imonth,nummth
      COMMON /RANS/ CPDRY,CPH2O,RDRY,FMER(0:NB),FMSEN(0:NB),FMLAT(0:NB),
     1              FMOC(0:NB),FZON(NB,0:nl),FZSEN(0:nl),FZLAT(0:nl),

      double precision CPDRY,CPH2O,RDRY,FMER,FMSEN,FMLAT,FMOC,FZON,
     1                 FZSEN,FZLAT,RML,RSOIL         
      COMMON /ALBE/ FO(NB,0:nl),FCLD(NB,0:nl),FI(NB,0:nl),ZEN(NB),

      double precision FO,FCLD,FI,ZEN,CZEN,ALBL,

      double precision AREA_LAND,AREA_SHELF             
      COMMON /ORBIT/ EXC,OBL,SMA,SUNEA,XLSPER
      double precision EXC,OBL,SMA,SUNEA,XLSPER
      COMMON /TEMPS/ time0,dn,imth
      double precision time0,dn
      integer imth
      common /CO2/ pco2
      double precision pco2  
  
      DOUBLE PRECISION y(neq),drv(neq),time,timedif
      double precision A0,B0,TCELS,TEM0,puis_p,puis_t,sold(nb)
      integer i,j,ieq

      timedif = time-time0

      if (timedif.gt.0) then
        nyr = int(timedif)
        dn = float(nday)*(timedif-float(nyr))
      else
        nyr = int(timedif-1.)
        dn = float(nday)*(timedif-float(nyr))
      endif
                  
      CALL SOLDAY(XLAT,DN,SOLD,ZEN,CZEN)
      
      TEMGL=0
      puis_p=GRAV/(GAMA*RDRY)
      puis_t=RDRY/CPDRY
 
      DO i=1,NB
       DO j=1,nl
        ieq=(i-1)*nl+j
        temp(i,j) = y(ieq)
        sol(i,j) = sold(i)
        elev(i,j)=max(0.,(1.-fo(i,j))*ELEVL(I,J))
        TEM0=temp(i,j)+GAMA*ELEV(I,J)
        ps(i,j) = PRE*(TEMP(I,j)/TEM0)**puis_p
        ptem(i,j) = temp(i,j)
     &      *(610./ps(i,j))**puis_t				!updated
       end do
       temp(i,0)=0.
       sol(i,0)=0.
       elev(i,0)=0.
       ps(i,0)=0.
       ptem(i,0)=0.
       DO j=1,nl
        TEMGL = TEMGL + temp(i,j)*SURF(i,j)
        temp(i,0) = temp(i,0) + temp(i,j)*SURF(i,j)/SURF(i,0)
        sol(i,0) = sol(i,0) + sol(i,j)*SURF(i,j)/SURF(i,0)
        elev(i,0) = elev(i,0) + elev(i,j)*SURF(i,j)/SURF(i,0)
        ps(i,0) = ps(i,0) + ps(i,j)*SURF(i,j)/SURF(i,0)
        ptem(i,0) = ptem(i,0) + ptem(i,j)*SURF(i,j)/SURF(i,0)
       end do       
      END DO
      
      IF(PCO2.GE.280.)THEN
         A0=A1+A2*DLOG(PCO2/280.)
         B0=B1+B2*DLOG(PCO2/280.)
        ELSE
         A0=A1+A2*((PCO2/280.)-1)
         B0=B1+B2*((PCO2/280.)-1)
      ENDIF

      CALL ALBEDO_BUD
      CALL DFLMER
      CALL DFLZON
      
      SOLGL = 0.
      ALBGL = 0.
      ALBSGL = 0.
      DO I=1,NB
       do j=1,nl
        TCELS=temp(i,j)-273.15
        FIR(i,j)=A0+B0*TCELS-(CC0+DD0*TCELS)*FCLD(i,j)
       end do
       fir(i,0)=0.
       do j=1,nl
        SOLGL = SOLGL + xlum*sol(i,j)*SURF(i,j)
        ALBGL = ALBGL + alb(i,j)*xlum*sol(i,j)*SURF(i,j)
        ALBSGL = ALBSGL + alb_surf(i,j)*xlum*sol(i,j)*SURF(i,j)
        fir(i,0) = fir(i,0) + fir(i,j)*SURF(i,j)/SURF(i,0)
       end do
      END DO
      ALBGL = ALBGL/SOLGL
      ALBSGL = ALBSGL/SOLGL

c   air temperature in latitude bands
      do i=1,nb
       do j=1,nl
        ieq=(i-1)*nl+j
        if (SURF(i,j) .gt. 1.d-10) then					! Updated
         drv(ieq)=(XLUM*SOL(i,j)*(1.-ALB(i,j))-FIR(i,j)+DFMER(i,j)
     &      +DFZON(i,j))*3.15E+7/CHEAT(i,j)
        else
         drv(ieq)=0.0d0
        endif
c        drv(ieq)=(XLUM*SOL(i,j)*(1.-ALB(i,j))-FIR(i,j)+DFMER(i,j))
c     &       *3.15E+7/CHEAT(i,j)
       end do
      end do 
      
c      write(*,*)'dn: ',dn
c      write(*,*)'temp: ',temp
c      write(*,*)'zen: ',zen
c      write(*,*)'fsnw: ',fsnw
c      write(*,*)'fi: ',fi
c      write(*,*)'sol: ',sol
c      write(*,*)'alb: ',alb
c      write(*,*)'fir: ',fir
c      write(*,*)'dfmer: ',dfmer
c      write(*,*)'cheat: ',cheat
      
c      stop
           
      return
      end

c     ----------------------------------------
      SUBROUTINE SOLDAY(XLAT,DN,SOLD,ZEN,CZEN)
C     ----------------------------------------
      implicit none
      integer nday,nb,nl
      PARAMETER (nday=668,NB=18,nl=2)
      COMMON /GEN/ PI,CHEAT(NB,0:nl),GRAV,REARTH,SEARTH,ITER
      double precision PI,CHEAT,GRAV,REARTH,SEARTH
      integer ITER
      COMMON /ORBIT/ EXC,OBL,SMA,SUNEA,XLSPER
      double precision EXC,OBL,SMA,SUNEA,XLSPER
      COMMON /FUN/ SDEL,CDEL,SPHI,CPHI,CH0,H0   
      double precision SDEL,CDEL,SPHI,CPHI,CH0,H0
      double precision SOLD(NB),XLAT(NB)
      double precision ZEN(NB),CZEN(NB)
      double precision DN
      integer i
      double precision obl_rad,xlsper_rad,slg,th,sth,xls,theta,
     &                 rm,cte,cz0
       
      obl_rad = obl*pi/180.d+0
      xlsper_rad = xlsper*pi/180.d+0
             
      do i = 1, NB
c=======================================================================
c   calculates necessary angles:
c   sdel  = sinus of solar declination;
c   h0    = solar hour angle at sunset
c=======================================================================
    
         slg   = (2. * pi / float(nday)) * (dn - 79.092)
         if(slg .lt. 0.) slg = 2. * pi + slg
         th    = slg - xlsper_rad
         sth   = dsin(th)
         xls   = slg + 2.*sth*exc + 2.5*sth*dcos(th)*exc*exc
     &        + (3.*sth - 13.*sth*sth*sth/3.) *exc*exc*exc
         theta = xls - xlsper_rad
         rm    = (1. - exc*exc) / (1. + exc*dcos(theta))
         sdel  = dsin(obl_rad) * dsin(xls)
	 
         cdel  = dsqrt(1.-sdel*sdel)
         cte   = sunea / rm * rm

         SPHI=XLAT(I)
         CPHI=DSQRT(1.-SPHI*SPHI)

         if(cphi .le. 1.d-20) cphi = 1.d-20

         ch0   = -sphi * sdel / (cphi * cdel)
         if(ch0 .ge. 1.) then
               h0 = 0.
          else
               if(ch0 .le. -1) then
                  h0 = pi
               else
                  h0 = acos(ch0)
               endif
         endif
	

c=======================================================================
c   calculates  the mean cosine of the solar zenith angle.
c=======================================================================
     
          cz0 = (h0 * sphi * sdel + cphi * cdel * dsin(h0)) / pi      
          CZEN(I) = cz0
	  ZEN(I) = ACOS(CZEN(I))
	  sold(i) = cte*cz0
      end do
      
      return
      end      

      
c     -----------------      
      SUBROUTINE DFLMER
C     -----------------

      implicit none
      integer nb,nl
      PARAMETER (NB=18,nl=2)

      COMMON /GEN/ PI,CHEAT(NB,0:nl),GRAV,REARTH,SEARTH,ITER
      double precision PI,CHEAT,GRAV,REARTH,SEARTH
      integer ITER
      COMMON /DIFU/ DIFS(0:NB),DIFL(0:NB),DIFO(0:NB),DIFZ(0:NB)
      double precision DIFS,DIFL,DIFO,DIFZ
      COMMON /CLIM/ temp(NB,0:nl),SOL(NB,0:nl),ALB(NB,0:nl),FIR(NB,0:nl)
     1             ,DFMER(NB,0:nl),PHI(NB),XLAT(NB),A1,A2,B1,B2
     2             ,XLUM,TEMGL,ALBGL,SOLGL,ALBSGL,PRE,ES(NB,0:nl)
     3             ,PHI1(NB),PHI2(NB),CC0,DD0,ELEVL(NB,0:nl)
     4             ,ELEV(NB,0:nl),PTEM(NB,0:nl),PS(NB,0:nl),XL(NB,0:nl)
     5             ,DFZON(NB,0:nl),DIST(NB,0:nl),SURF(NB,0:nl),GAMA
      double precision temp,SOL,ALB,FIR,DFMER,PHI,XLAT,A1,A2,B1,B2
     1             ,XLUM,TEMGL,ALBGL,SOLGL,ALBSGL,PRE,ES
     2             ,PHI1,PHI2,CC0,DD0,ELEVL,ELEV,PTEM,PS,XL
     3             ,DFZON,DIST,SURF,GAMA
      COMMON /ALBE/ FO(NB,0:nl),FCLD(NB,0:nl),FI(NB,0:nl),ZEN(NB),
     1              CZEN(NB),ALBL(NB,0:nl),ALICE,ALOC,ALC0,ALC1,
     2              ALCR(NB,0:nl),ALCY(NB,0:nl),alb_surf(nb,0:nl),
     3              FSNW(NB,0:nl),ALSNW,TSUP_ICE,TINF_ICE,TSUP_SNW,
     4              TINF_SNW
      double precision FO,FCLD,FI,ZEN,CZEN,ALBL,
     1    ALICE,ALOC,ALC0,ALC1,ALCR,ALCY,alb_surf,FSNW,ALSNW,TSUP_ICE,
     2    TINF_ICE,TSUP_SNW,TINF_SNW 
      COMMON /RANS/ CPDRY,CPH2O,RDRY,FMER(0:NB),FMSEN(0:NB),FMLAT(0:NB),
     1              FMOC(0:NB),FZON(NB,0:nl),FZSEN(0:nl),FZLAT(0:nl),
     1              RML,RSOIL
      double precision CPDRY,CPH2O,RDRY,FMER,FMSEN,FMLAT,FMOC,FZON,
     1                 FZSEN,FZLAT,RML,RSOIL
      COMMON /ORBIT/ EXC,OBL,SMA,SUNEA,XLSPER
      double precision EXC,OBL,SMA,SUNEA,XLSPER

      integer i,j
      double precision HSEN(NB),HLAT(NB),HOC(NB),XL1,XL2,FACGEO,DXL2,
     &                 FAC2,SF2,SF1,FOM

      DO I=1,NB
c       HSEN(I)=(PRE*CPDRY/GRAV+1.06*ES(I,0)*(CPH2O-CPDRY))*temp(I,0)
c   PRE*CPDRY*TEMP/GRAV = sensible heat storage in atmospheric column (J m-2)
       HSEN(I)=(PRE*CPDRY/GRAV+1.06*ES(I,0)*(CPH2O-CPDRY))
     &      *ptem(I,0)
       HLAT(I)=XL(I,0)*1.06*ES(I,0)
       HOC(I)=RML*temp(I,0)
      END DO
      FMSEN(0)=0.
      FMLAT(0)=0.
      FMOC(0)=0.
      FMER(0)=0.
      XL1=-1.
      FACGEO=2.*PI*REARTH*REARTH
      DO I=1,NB-1
       XL2=SIN(PHI2(I)*PI/180.)
       DXL2=XLAT(I+1)-XLAT(I)
       FAC2=2.*PI*(1.-XL2*XL2)
       SF2=XLAT(I+1)-XL2
       SF1=XL2-XLAT(I)
       FOM=(FO(I,0)*SF1+FO(I+1,0)*SF2)/(SF1+SF2)
       FMSEN(I)=FAC2*DIFS(I)*(HSEN(I+1)-HSEN(I))/DXL2
       FMLAT(I)=FAC2*DIFL(I)*(HLAT(I+1)-HLAT(I))/DXL2
       if(fi(i,0).gt.0.)then
         fmoc(i)=(FAC2*FOM*DIFO(I)*(HOC(I+1)-HOC(I))/DXL2)*(1-fi(i,0))
       else
         FMOC(I)=FAC2*FOM*DIFO(I)*(HOC(I+1)-HOC(I))/DXL2
       endif 
       FMER(I)=FMSEN(I)+FMLAT(I)+FMOC(I)
       DFMER(I,0)=(FMER(I)-FMER(I-1))/((XL2-XL1)*FACGEO)
       XL1=XL2
      END DO
      FMSEN(NB)=0.
      FMLAT(NB)=0.
      FMOC(NB)=0.
      FMER(NB)=0.
      DFMER(NB,0)=-FMER(NB-1)/((1.-XL1)*FACGEO)

      DO I=1,NB
       DO J=1,nl
        DFMER(I,J)=DFMER(I,0)
       END DO
      END DO

      RETURN
      END

c     -----------------      
      SUBROUTINE DFLZON
C     -----------------

      implicit none
      integer nb,nl
      PARAMETER (NB=18,nl=2)

      COMMON /GEN/ PI,CHEAT(NB,0:nl),GRAV,REARTH,SEARTH,ITER
      double precision PI,CHEAT,GRAV,REARTH,SEARTH
      integer ITER
      COMMON /DIFU/ DIFS(0:NB),DIFL(0:NB),DIFO(0:NB),DIFZ(0:NB)
      double precision DIFS,DIFL,DIFO,DIFZ
      COMMON /CLIM/ temp(NB,0:nl),SOL(NB,0:nl),ALB(NB,0:nl),FIR(NB,0:nl)
     1             ,DFMER(NB,0:nl),PHI(NB),XLAT(NB),A1,A2,B1,B2
     2             ,XLUM,TEMGL,ALBGL,SOLGL,ALBSGL,PRE,ES(NB,0:nl)
     3             ,PHI1(NB),PHI2(NB),CC0,DD0,ELEVL(NB,0:nl)
     4             ,ELEV(NB,0:nl),PTEM(NB,0:nl),PS(NB,0:nl),XL(NB,0:nl)
     5             ,DFZON(NB,0:nl),DIST(NB,0:nl),SURF(NB,0:nl),GAMA
      double precision temp,SOL,ALB,FIR,DFMER,PHI,XLAT,A1,A2,B1,B2
     1             ,XLUM,TEMGL,ALBGL,SOLGL,ALBSGL,PRE,ES
     2             ,PHI1,PHI2,CC0,DD0,ELEVL,ELEV,PTEM,PS,XL
     3             ,DFZON,DIST,SURF,GAMA
      COMMON /ALBE/ FO(NB,0:nl),FCLD(NB,0:nl),FI(NB,0:nl),ZEN(NB),
     1              CZEN(NB),ALBL(NB,0:nl),ALICE,ALOC,ALC0,ALC1,
     2              ALCR(NB,0:nl),ALCY(NB,0:nl),alb_surf(nb,0:nl),
     3              FSNW(NB,0:nl),ALSNW,TSUP_ICE,TINF_ICE,TSUP_SNW,
     4              TINF_SNW
      double precision FO,FCLD,FI,ZEN,CZEN,ALBL,
     1    ALICE,ALOC,ALC0,ALC1,ALCR,ALCY,alb_surf,FSNW,ALSNW,TSUP_ICE,
     2    TINF_ICE,TSUP_SNW,TINF_SNW 
      COMMON /RANS/ CPDRY,CPH2O,RDRY,FMER(0:NB),FMSEN(0:NB),FMLAT(0:NB),
     1              FMOC(0:NB),FZON(NB,0:nl),FZSEN(0:nl),FZLAT(0:nl),
     1              RML,RSOIL
      double precision CPDRY,CPH2O,RDRY,FMER,FMSEN,FMLAT,FMOC,FZON,
     1                 FZSEN,FZLAT,RML,RSOIL
      COMMON /ORBIT/ EXC,OBL,SMA,SUNEA,XLSPER
      double precision EXC,OBL,SMA,SUNEA,XLSPER

      integer i,j,jn
      double precision HZSEN(nl),HZLAT(nl),PARLGT,FACGEO,DIST1

c zonal diffusion coefficient (m2 s-1)
c      DIFZ(*) = 8.e+4 * 100.

      DO I=1,NB

       do j = 1, nl
       HZSEN(j)=(PRE*CPDRY/GRAV+1.06*ES(I,j)*(CPH2O-CPDRY))*ptem(I,j)
       HZLAT(j)=XL(I,j)*1.06*ES(I,j)
       end do
      
       PARLGT=2.*PI*REARTH*DCOS(PHI(I)*PI/180.)
       DO j=1,nl
        if (j.eq.nl) then 
          jn=1
        else
          jn=j+1
        endif
        if ((surf(i,j).ge.1.d-10).and.(surf(i,jn).ge.1.d-10)) then       
         DIST(i,j)=0.5*(SURF(I,J)+SURF(I,JN))*PARLGT/SURF(I,0)
         FZSEN(j)=-DIFZ(i)*(HZSEN(JN)-HZSEN(J))/DIST(i,j)
         FZLAT(j)=-DIFZ(i)*(HZLAT(JN)-HZLAT(J))/DIST(i,j)
         FZON(I,j)=FZSEN(J)+FZLAT(J)
        else
         DIST(I,J)=0.
         FZSEN(J)=0.
         FZLAT(J)=0.
         FZON(I,j)=0.
        endif
       END DO
       FZSEN(0)=FZSEN(NL)
       FZLAT(0)=FZLAT(NL)
       FZON(I,0)=FZON(I,NL)

       DO j = 1, nl
        DIST1=SURF(I,J)*PARLGT/SURF(I,0)
        if (surf(i,j).ge.1.d-10) then
          DFZON(I,J)=-(FZON(I,j)-FZON(I,j-1))/DIST1
        else
          DFZON(I,J)=0.
        endif
       END DO

       DFZON(I,0)=0.
       DO J=1,nl
         DFZON(I,0)=DFZON(I,0)+DFZON(I,J)*SURF(I,J)/SURF(I,0)
       END DO

      END DO

      RETURN
      END

c     -----------------------
      SUBROUTINE ALBEDO_BUD
C     -----------------------

      implicit none
      integer nb,nl
      PARAMETER (NB=18,nl=2)

      COMMON /CLIM/ temp(NB,0:nl),SOL(NB,0:nl),ALB(NB,0:nl),FIR(NB,0:nl)
     1             ,DFMER(NB,0:nl),PHI(NB),XLAT(NB),A1,A2,B1,B2
     2             ,XLUM,TEMGL,ALBGL,SOLGL,ALBSGL,PRE,ES(NB,0:nl)
     3             ,PHI1(NB),PHI2(NB),CC0,DD0,ELEVL(NB,0:nl)
     4             ,ELEV(NB,0:nl),PTEM(NB,0:nl),PS(NB,0:nl),XL(NB,0:nl)
     5             ,DFZON(NB,0:nl),DIST(NB,0:nl),SURF(NB,0:nl),GAMA
      double precision temp,SOL,ALB,FIR,DFMER,PHI,XLAT,A1,A2,B1,B2
     1             ,XLUM,TEMGL,ALBGL,SOLGL,ALBSGL,PRE,ES
     2             ,PHI1,PHI2,CC0,DD0,ELEVL,ELEV,PTEM,PS,XL
     3             ,DFZON,DIST,SURF,GAMA
      COMMON /GEN/ PI,CHEAT(NB,0:nl),GRAV,REARTH,SEARTH,ITER
      double precision PI,CHEAT,GRAV,REARTH,SEARTH
      integer ITER
      COMMON /DIFU/ DIFS(0:NB),DIFL(0:NB),DIFO(0:NB),DIFZ(0:NB)
      double precision DIFS,DIFL,DIFO,DIFZ
      COMMON /ALBE/ FO(NB,0:nl),FCLD(NB,0:nl),FI(NB,0:nl),ZEN(NB),
     1              CZEN(NB),ALBL(NB,0:nl),ALICE,ALOC,ALC0,ALC1,
     2              ALCR(NB,0:nl),ALCY(NB,0:nl),alb_surf(nb,0:nl),
     3              FSNW(NB,0:nl),ALSNW,TSUP_ICE,TINF_ICE,TSUP_SNW,
     4              TINF_SNW
      double precision FO,FCLD,FI,ZEN,CZEN,ALBL,
     1    ALICE,ALOC,ALC0,ALC1,ALCR,ALCY,alb_surf,FSNW,ALSNW,TSUP_ICE,
     2    TINF_ICE,TSUP_SNW,TINF_SNW 
      COMMON /ORBIT/ EXC,OBL,SMA,SUNEA,XLSPER
      double precision EXC,OBL,SMA,SUNEA,XLSPER

      integer i,j
      double precision alb_cloud,alb_atm,alb_ocean


      alb_cloud = ALC0   ! 0.44
      alb_atm = ALC1     ! 0.04
      alb_ocean = ALOC   ! 0.065
c      write(*,*) 'tsup_ice: ', tsup_ice
c      write(*,*) 'tinf_ice: ', tinf_ice
c      write(*,*) 'tsup_snw: ', tsup_snw
c      write(*,*) 'tinf_snw: ', tinf_snw
            
      do i = 1, nb
       do j = 1, nl
	! Considering the moisture parameter to be zero for dry conditions (Mars). 		!Updated      	
	XL(i,j)=0.
       	ES(i,j)=0.  
        IF(temp(i,j).GT.TSUP_ICE)THEN
           FI(i,j)=0.
         ELSEIF(temp(i,j).GT.TINF_ICE)THEN
           FI(i,j)=(TSUP_ICE-temp(i,j))/(TSUP_ICE-TINF_ICE)
         ELSE
           FI(i,j)=1.
        ENDIF
        IF(temp(i,j).GE.TSUP_SNW)THEN
          FSNW(i,j) = 0.
        ELSEIF(temp(i,j).gt.TINF_SNW)THEN
	  FSNW(i,j) = (TSUP_SNW-temp(i,j))/(TSUP_SNW-TINF_SNW)
        ELSE
          FSNW(i,j) = 1.
        ENDIF
        ALCY(i,j) =(alb_atm/2.)+((1.-(alb_atm/2.))**2.)*alb_cloud
        alb_surf(i,j) = (1.-fo(i,j))*(1.-fsnw(i,j))*ALBL(i,j)      
     &             + (1.-fo(i,j))*fsnw(i,j)*alsnw
     &             + fo(i,j)*(1.-fi(i,j))*alb_ocean      
     &             + fo(i,j)*fi(i,j)*alice 
        ALCR(i,j) = alb_atm+((1-alb_atm)**2.)*alb_surf(i,j)
        ALB(i,j)=FCLD(i,j)*ALCY(i,j)+(1.-FCLD(i,j))*ALCR(i,j)
       end do

       xl(i,0)=0.
       es(i,0)=0.
       fi(i,0)=0.
       fsnw(i,0)=0.
       alcy(i,0)=0.
       alb_surf(i,0)=0.
       alcr(i,0)=0.
       alb(i,0)=0.
       do j = 1, nl
        xl(i,0) = xl(i,0)+xl(i,j)*SURF(i,j)/SURF(i,0)
        es(i,0) = es(i,0)+es(i,j)*SURF(i,j)/SURF(i,0)
        if (fo(i,0).gt.1.d-6) then 
          fi(i,0) = fi(i,0)+fi(i,j)*fo(i,j)*SURF(i,j)
     &              /(fo(i,0)*SURF(i,0))
        else
          fi(i,0)= 0.
        endif
        if (fo(i,0).lt.0.999999) then
          fsnw(i,0) = fsnw(i,0)+fsnw(i,j)*(1-fo(i,j))*SURF(i,j)
     &              /((1-fo(i,0))*SURF(i,0))
        else
          fsnw(i,0) = 0.
        endif
        alcy(i,0) = alcy(i,0)+alcy(i,j)*SURF(i,j)/SURF(i,0)
        alb_surf(i,0) = alb_surf(i,0)+alb_surf(i,j)*SURF(i,j)/SURF(i,0)
        alcr(i,0) = alcr(i,0)+alcr(i,j)*SURF(i,j)/SURF(i,0)
        alb(i,0) = alb(i,0)+alb(i,j)*SURF(i,j)/SURF(i,0)
       end do
      end do
      
c      write(*,*)'temp: ',temp
c      write(*,*)'fsnw: ',fsnw
c      write(*,*)'fi: ',fi
c      write(*,*)'alb: ',alb             
      
      RETURN
      END
      
      



cccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccc
c
c  SUBROUTINES OF THE DIFFERENTIAL EQUATION SOLVER: RUNGE-KUTTA 
c           WITH VARIABLE TIME STEP
c 
cccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccccc

c     ******************************************************************
      SUBROUTINE ode(tbegin,tend,y,step_prt,step_min,step_max,eps,nyprt)
c     ******************************************************************

      implicit none
      integer nb,neq,nl
      DOUBLE PRECISION ZERO
      PARAMETER (nb=18,nl=2,neq=nb*nl,ZERO=0.D-40)
      DOUBLE PRECISION y(neq),drv(neq)
      DOUBLE PRECISION tbegin,tend,step_prt,step_min,step_max,eps
      integer nyprt
      integer NOK,NBAD
      double precision x1,x2,h1,hprt,hmin,hmax,TIME

      x1 = tbegin
      call givedrv(tbegin,y,drv)
      
      h1 = 0.1
      hprt = dabs(step_prt)

      call printing(tbegin,y,drv,hprt)
c      call write_ebm(tbegin)
      
      hmin = dabs(step_min)

1     continue

      if (tend.ge.tbegin) then      
        x2 = x1+hprt
        if(x2.gt.tend)x2=tend
        h1 = min(dabs(h1),x2-x1)
      else
        x2 = x1-hprt
        if(x2.lt.tend)x2=tend
        h1 = -min(dabs(h1),x1-x2)
      endif

      HMAX = dabs(step_max)
      call ODEINT(y,drv,X1,X2,EPS,H1,HMIN,HMAX,NOK,NBAD)
      time = x2
      call givedrv(time,y,drv)
      call printing(time,y,drv,h1)
      if ((tend-time).le.float(nyprt)) call write_ebm(time)
      x1 = x2

      IF ((time-tend)*(tend-tbegin).LT.ZERO) GOTO 1

      return
      end

c     ******************************************************************
      SUBROUTINE ODEINT(YSTART,DYSTART,X1,X2,EPS,H1,HMIN,HMAX,NOK,NBAD)
c     ******************************************************************

      implicit none
      integer nb,neq,nl,maxstp
      double precision TWO,ZERO,TINY
      PARAMETER (MAXSTP=50000000,nb=18,nl=2,neq=nb*nl					Updated!
     &          ,TWO=2.0,ZERO=0.D-40,TINY=0.1)
      !Considering the MAXSTP as per the Martian conditions	
      COMMON /PATH/ KMAX,KOUNT,DXSAV,XP(200),YP(10,200)
      integer KMAX,KOUNT
      double precision DXSAV,XP,YP
      COMMON /DEBG/ kdebug
      integer kdebug
      DOUBLE PRECISION YSTART(neq),DYSTART(neq),YSCAL(neq),Y(neq)
     &,DYDX(neq)

      integer NOK,NBAD
      double precision x1,x2,eps,h1,hmin,hmax

      integer i,n_min,NSTP
      double precision X,H,XSAV,hdid,hnext
      
      n_min = 0
      X=X1
      H=SIGN(H1,X2-X1)
      NOK=0
      NBAD=0
      KOUNT=0
      DO I=1,neq
        Y(I)=YSTART(I)
        DYDX(I)=DYSTART(I)
      END DO
      IF (KMAX .GT. 0) XSAV=X-DXSAV*TWO
      DO NSTP=1,MAXSTP
c        write(7,'(20(1x,D16.9))')HDID,X,Y
        CALL givedrv(X,Y,DYDX)
	if (kdebug.eq.1) call write_ebm(X)
        DO I=1,neq
          YSCAL(I)=DABS(Y(I))+DABS(H*DYDX(I))+TINY
        END DO
        IF(KMAX.GT.0)THEN
          IF(DABS(X-XSAV).GT.DABS(DXSAV)) THEN
            IF(KOUNT.LT.KMAX-1)THEN
              KOUNT=KOUNT+1
              XP(KOUNT)=X
              DO I=1,neq
                YP(I,KOUNT)=Y(I)
              END DO
              XSAV=X
            ENDIF
          ENDIF
        ENDIF
        IF((X+H-X2)*(X+H-X1).GT.ZERO) H=X2-X
        CALL RKQC(Y,DYDX,X,H,EPS,YSCAL,HDID,HNEXT,HMAX)			Updated!
        IF (DABS (HDID) .LT.1.0d-7) THEN
            HDID = SIGN(1.0d-7, X2 - X1)
            HNEXT = SIGN(1.0d-7, X2 - X1)
        ENDIF
        IF(HDID.EQ.H)THEN
          NOK=NOK+1
        ELSE
          NBAD=NBAD+1
        ENDIF
        IF((X-X2)*(X2-X1).GE.ZERO)THEN
          DO I=1,neq
            YSTART(I)=Y(I)
            DYSTART(I)=DYDX(I)
          END DO
          IF(KMAX.NE.0)THEN
            KOUNT=KOUNT+1
            XP(KOUNT)=X
            DO I=1,neq
              YP(I,KOUNT)=Y(I)
            END DO
          ENDIF
          H1 = HDID
c	  rewind(7)
          RETURN
        ENDIF
        IF(DABS(HNEXT).LT.HMIN) then
	 if (n_min.gt.200) then
           WRITE(7,*) 'Stepsize smaller than minimum'
           WRITE(7,*) '      Current stepsize: ',HDID
           WRITE(7,*) '      Next stepsize:    ',HNEXT
           WRITE(7,*) '      Minimum stepsize: ',HMIN
           STOP
	 endif
	 n_min = n_min+1
        ENDIF
        H=HNEXT
      END DO
      write(7,*)'Too many steps in ODINT.'
      STOP
      RETURN
      END

c     ******************************************************************
      SUBROUTINE RKQC(Y,DYDX,X,HTRY,EPS,YSCAL,HDID,HNEXT,HMAX)
c     ******************************************************************

      implicit none
      integer nb,neq,nl
      double precision FCOR,ONE,SAFETY,ERRCON
      PARAMETER (nb=18,nl=2,neq=nb*nl
     &          ,FCOR=.0666666667,ONE=1.,SAFETY=0.9,ERRCON=6.E-4)    !ERRCON=6.E-4
      DOUBLE PRECISION Y(neq),DYDX(neq),YSCAL(neq),YTEMP(neq)
     &         ,YSAV(neq),DYSAV(neq)
      COMMON /DEBG/ kdebug
      integer kdebug

      double precision X,HTRY,EPS,HDID,HNEXT,HMAX

      integer i
      double precision PGROW,PSHRNK,XSAV,H,ERRMAX,HH
           
      PGROW=-0.20
      PSHRNK=-0.25
      XSAV=X
      DO I=1,neq
        YSAV(I)=Y(I)
        DYSAV(I)=DYDX(I)
      END DO
      H=HTRY
1     HH=0.5*H
      CALL RK4(YSAV,DYSAV,XSAV,HH,YTEMP)
      X=XSAV+HH
      CALL givedrv(X,YTEMP,DYDX)
      if (kdebug.eq.1) call write_ebm(X)
      CALL RK4(YTEMP,DYDX,X,HH,Y)
      X=XSAV+H
      IF(X.EQ.XSAV)THEN
        WRITE(7,*) 'Stepsize not significant in RKQC.'
        WRITE(7,*)' H=',H
        WRITE(7,*)' X=',X
        WRITE(7,*)' XSAV=',XSAV
        STOP
      ENDIF
      CALL RK4(YSAV,DYSAV,XSAV,H,YTEMP)
      ERRMAX=0.
      DO I=1,neq
        YTEMP(I)=Y(I)-YTEMP(I)
        ERRMAX=MAX(ERRMAX,DABS(YTEMP(I)/YSCAL(I)))
      END DO
      ERRMAX=ERRMAX/EPS
      IF(ERRMAX.GT.ONE) THEN
c        hanci=H
        H=SAFETY*H*(ERRMAX**PSHRNK)
        GOTO 1
      ELSE
        HDID=H
        IF(ERRMAX.GT.ERRCON)THEN
          HNEXT=SAFETY*H*(ERRMAX**PGROW)
        ELSE
          HNEXT=4.*H
        ENDIF
        IF (HNEXT.GE.HMAX) HNEXT=HMAX
      ENDIF
      DO I=1,neq
        Y(I)=Y(I)+YTEMP(I)*FCOR
      END DO
      RETURN
      END


c     ******************************************************************
      SUBROUTINE RK4(Y,DYDX,X,H,YOUT)
c     ******************************************************************

      implicit none
      integer nb,neq,nl     
      PARAMETER (nb=18,nl=2,neq=nb*nl)
      DOUBLE PRECISION Y(neq),DYDX(neq),YOUT(neq),YT(neq),DYT(neq)
     &,DYM(neq),X,H
      COMMON /DEBG/ kdebug
      integer kdebug 

      integer i
      double precision HH,H6,XH
     
      HH=H*0.5
      H6=H/6.
      XH=X+HH
      DO I=1,neq
        YT(I)=Y(I)+HH*DYDX(I)
      END DO
      CALL givedrv(XH,YT,DYT)
      if (kdebug.eq.1) call write_ebm(XH)
      DO I=1,neq
        YT(I)=Y(I)+HH*DYT(I)
      END DO
      CALL givedrv(XH,YT,DYM)
      if (kdebug.eq.1) call write_ebm(XH)
      DO I=1,neq
        YT(I)=Y(I)+H*DYM(I)
        DYM(I)=DYT(I)+DYM(I)
      END DO
      CALL givedrv(X+H,YT,DYT)
      if (kdebug.eq.1) call write_ebm(X+H)
      DO I=1,neq
        YOUT(I)=Y(I)+H6*(DYDX(I)+DYT(I)+2.*DYM(I))
      END DO

      RETURN
      END

     &             -10.1280191D0,   10.0289441D0,  -10.0034259D0/

      data mvrate /31.609974D0, 32.620504D0, 24.172203D0,               &
     &              0.636717D0, 31.983787D0,  3.138886D0,               &
     &             30.973257D0, 44.828336D0,  0.991874D0,               &

     &              7.437771D0, 18.829299D0,  9.500642D0,               &

     &              1.647247D0, 10.884985D0,  5.610937D0,               &
     &             12.658184D0,  1.010530D0,  1.983748D0,               &
     &             14.023871D0,  0.560178D0,  1.273434D0,               &
     &             12.021467D0, 62.583231D0, 63.593761D0,               &
     &             76.438310D0,  4.280910D0, 13.218362D0,               &
     &             17.818769D0,  8.359495D0, 56.792707D0,               &

     &             45.815258D0, 12.002811D0, 75.278220D0,               &

     &              9.237738D0, 49.747842D0,  2.147012D0,               &
     &              1.196895D0,  2.133898D0,  0.173168D0/

      data mvphas /251.9025D0, 280.8325D0, 128.3057D0,                  &
     &             348.1074D0, 292.7252D0, 165.1686D0,                  &
     &             263.7951D0,  15.3747D0,  58.5749D0,                  &
     &              40.8226D0, 308.4258D0, 240.0099D0,                  &
     &             222.9725D0, 106.5937D0, 114.5182D0,                  &
     &             268.7809D0, 279.6869D0,  39.6448D0,                  &
     &             126.4108D0, 291.5795D0, 307.2848D0,                  &
     &              18.9300D0, 273.7596D0, 143.8050D0,                  &
     &             191.8927D0, 125.5237D0, 172.7351D0,                  &
     &             316.7998D0, 319.6024D0,  69.7526D0,                  &

     &              17.0374D0, 155.3409D0,  94.1709D0,                  &
     &             221.1120D0,  28.9300D0, 117.1498D0,                  &
     &             320.5095D0, 262.3602D0, 336.2148D0,                  &
     &             233.0046D0, 155.6977D0, 184.6277D0,                  &
     &             267.2772D0,  78.9281D0, 123.4722D0,                  &
     &             188.7132D0, 180.1364D0,  49.1382D0,                  &
     &             152.5268D0,  98.2198D0,  97.4808D0,                  &
     &             221.5376D0, 168.2438D0, 161.1199D0,                  &
     &              55.0196D0, 262.6495D0, 200.3284D0,                  &
     &             201.6651D0, 294.6547D0,  99.8233D0,                  &
     &             213.5577D0, 154.1631D0, 232.7153D0,                  &
     &             138.3034D0, 204.6609D0, 106.5938D0,                  &
     &             250.4676D0, 332.3345D0,  27.3039D0/

c     Local variables
c     ---------------
      integer i        ! Index for series summations
      real :: obsum    ! Obliquity series summation
      real :: cossum   ! Cosine series summation for eccentricity/fvelp
      real :: sinsum   ! Sine series summation for eccentricity/fvelp
      real :: fvelp    ! Fixed vernal equinox longitude of perihelion
      real :: mvsum    ! mvelp series summation
      real :: beta     ! Intermediate argument for lambm0
      real :: years    ! Years to time of interest (negative = past;
c                      ! positive = future)
      real :: exc2   ! eccentricity squared
      real :: exc3   ! eccentricity cubed
CC      real :: pi       ! pi 

c radinp and algorithms below will need a degrees to radians conversion
c factor.

      pi     =  4.*atan(1.)
      degrad = pi/180.

c Check for flag to use input orbit parameters
c        i_AD = myear  !!!!!! REMPLACER PAR nyear2 si Holocene
c        if(imig.eq.1) i_AD = 1950 - nyear2

      if ( i_AD .eq. ORB_NOT_YEAR_BASED ) then

c Check input obliq, eccen, and mvelp to ensure reasonable

         if( obl .eq. ORB_UNDEF_REAL )then
          if ( lg_p ) then
            write(*,*)'(orb_params) Have to specify orbital parameters:'
            write(*,*) 'Either set: '                                   &
     &                ,'i_AD, OR [obl, exc, and xlsper]:'
            write(*,*)'i_AD is the year to simulate the orbit for ' 
     &                ,'(ie. 1950): '
            write(*,*)'obl, exc, xlsper specify the orbit directly:'
            write(*,*)'The AMIP II settings (for a 1995 orbit) are: '
            write(*,*)' obl = 23.4441'
            write(*,*)' exc = 0.016715'
            write(*,*)' xlsper = 102.7'
          end if
          stop 999
         else if ( lg_p ) then
           write(*,*)'(orb_params) Use input orbital parameters: '
         end if
         if( (obl.lt.ORB_OBLIQ_MIN).or.(obl.gt.ORB_OBLIQ_MAX) ) then
          if ( lg_p ) then
             write(*,*) '(orb_params): Input obliquity unreasonable: '  &
     &                  ,obl
          end if
          stop 999
         end if
         if( (exc.lt.ORB_ECCEN_MIN).or.(exc.gt.ORB_ECCEN_MAX) ) then
          if ( lg_p ) then
            write(*,*) '(orb_params): Input eccentricity unreasonable: '&
     &                 ,exc
          end if
          stop 999
         end if
         if( (xlsper.lt.ORB_MVELP_MIN).or.
     &             (xlsper.gt.ORB_MVELP_MAX) ) then
          if ( lg_p ) then
         write(*,*)'(orb_params): Input mvelp unreasonable: ',xlsper
          end if
          stop 999
         end if
        exc2 = exc*exc
        exc3 = exc2*exc
      else

c Otherwise calculate based on years before present

        yb4_1950AD = 1950.0 - float(i_AD)
        if ( abs(yb4_1950AD) .gt. 1000000.0 )then
          if ( lg_p ) then
            write(*,*)'(orb_params) orbit only valid for years+-1000000'
            write(*,*)'(orb_params) Relative to 1950 AD'
            write(*,*)'(orb_params) # of years before 1950: ',yb4_1950AD
            write(*,*)'(orb_params) Year to simulate was  : ',i_AD
          end if
          stop 999
        end if


c The following calculates the earth's obliquity, orbital eccentricity
c (and various powers of it) and vernal equinox mean longitude of
c perihelion for years in the past (future = negative of years past),
c using constants (see parameter section) given in the program of:

c Berger, Andre.  1978  A Simple Algorithm to Compute Long-Term Variations
c of Daily Insolation.  Contribution 18, Institute of Astronomy and
c Geophysics, Universite Catholique de Louvain, Louvain-la-Neuve, Belgium.

c and formulas given in the paper (where less precise constants are also
c given):

c Berger, Andre.  1978.  Long-Term Variations of Daily Insolation and
c Quaternary Climatic Changes.  J. of the Atmo. Sci. 35:2362-2367

c The algorithm is valid only to 1,000,000 years past or hence.
c For a solution valid to 5-10 million years past see the above author.
c Algorithm below is better for years closer to present than is the
c 5-10 million year solution.

c Years to time of interest must be negative of years before present
c (1950) in formulas that follow.

        years = - yb4_1950AD

c In the summations below, cosine or sine arguments, which end up in
c degrees, must be converted to radians via multiplication by degrad.

c Summation of cosine series for obliquity (epsilon in Berger 1978) in
c degrees. Convert the amplitudes and rates, which are in arc seconds, into
c degrees via multiplication by psecdeg (arc seconds to degrees conversion
c factor).  For obliq, first term is Berger 1978's epsilon star; second
c term is series summation in degrees.
c
        obsum = 0.0
        do i = 1, poblen
          obsum = obsum +                                               &
     &           obamp(i)*psecdeg*cos((obrate(i)*psecdeg*years +        &
     &                                obphas(i))*degrad)
        end do
        obl = 23.320556 + obsum

c Summation of cosine and sine series for computation of eccentricity
c (eccen; e in Berger 1978) and fixed vernal equinox longitude of perihelion
c (fvelp; pi in Berger 1978), which is used for computation of moving vernal
c equinox longitude of perihelion.  Convert the rates, which are in arc
c seconds, into degrees via multiplication by psecdeg.

        cossum = 0.0
        do i = 1, pecclen
          cossum = cossum +                                             &
     &            ecamp(i)*cos((ecrate(i)*psecdeg*years +               &
     &                          ecphas(i))*degrad)
        end do

        sinsum = 0.0
        do i = 1, pecclen
          sinsum = sinsum +                                             &
     &            ecamp(i)*sin((ecrate(i)*psecdeg*years +               &
     &                          ecphas(i))*degrad)
        end do

c Use summations to calculate eccentricity

        exc2 = cossum*cossum + sinsum*sinsum
        exc = sqrt(exc2)
        exc3 = exc2*exc

c A series of cases for fvelp, which is in radians.

        if (abs(cossum) .le. 1.0E-8) then
          if (sinsum .eq. 0.0) then
            fvelp = 0.0
          else if (sinsum .lt. 0.0) then
            fvelp = 1.5*pi
          else if (sinsum .gt. 0.0) then
            fvelp = .5*pi
          endif
        else if (cossum .lt. 0.0) then
          fvelp = atan(sinsum/cossum) + pi
        else if (cossum .gt. 0.0) then
          if (sinsum .lt. 0.0) then
            fvelp = atan(sinsum/cossum) + 2.0*pi
          else
            fvelp = atan(sinsum/cossum)
          endif
        endif

c Summation of sine series for computation of moving vernal equinox longitude
c of perihelion (mvelp; omega bar in Berger 1978) in degrees.  For mvelp,
c first term is fvelp in degrees; second term is Berger 1978's psi bar times
c years and in degrees; third term is Berger 1978's zeta; fourth term is
c series summation in degrees.  Convert the amplitudes and rates, which are
c in arc seconds, into degrees via multiplication by psecdeg.  Series summation
c plus second and third terms constitute Berger 1978's psi, which is the
c general precession.

        mvsum = 0.0
        do i = 1, pmvelen
          mvsum = mvsum +                                               &
     &           mvamp(i)*psecdeg*sin((mvrate(i)*psecdeg*years +        &
     &                                mvphas(i))*degrad)
        end do
        xlsper = fvelp/degrad + 50.439273*psecdeg*years + 3.392506       &
     &  + mvsum

c Cases to make sure mvelp is between 0 and 360.

        do while (xlsper .lt. 0.0)
          xlsper = xlsper + 360.0
        end do
        do while (xlsper .ge. 360.0)
          xlsper = xlsper - 360.0
        end do
      end if  ! end of test on whether to calculate or use input orbital params

c Orbit needs the obliquity in radians

      oblr = obl*degrad

c 180 degrees must be added to mvelp since observations are made from the
c earth and the sun is considered (wrongly for the algorithm) to go around
c the earth. For a more graphic explanation see Appendix B in:

c A. Berger, M. Loutre and C. Tricot. 1993.  Insolation and Earth's Orbital
c Periods.  J. of Geophysical Research 98:10,341-10,362.

c Additionally, orbit will need this value in radians. So mvelp becomes
c mvelpp (mvelp plus pi)

      xlsperp = (xlsper + 180.)*degrad

c Set up an argument used several times in lambm0 calculation ahead.

      beta = sqrt(1. - exc2)

c The mean longitude at the vernal equinox (lambda m nought in Berger
c 1978; in radians) is calculated from the following formula given in
c Berger 1978.  At the vernal equinox the true longitude (lambda in Berger
c 1978) is 0.

      lambm0 = 2.*((.5*exc + .125*exc3)*(1. + beta)*sin(xlsperp)     
     &            - .25*exc2*(.5 + beta)*sin(2.*xlsperp)               
     &            + .125*exc3*(1./3. + beta)*sin(3.*xlsperp))
     
      xlsper= xlsper+180.
      if (xlsper.ge.360.)xlsper=xlsper-360.

c      if ( lg_p ) then
c        write(*,*) exc,obl,xlsper
        write(*,'(/," ****************************************")')
        write(*,'(" *     Computed Orbital Parameters      *")')
        write(*,'(" ****************************************")')
        write(*,'(" * Year AD           = ",i16  ," *")') i_AD
        write(*,'(" * Eccentricity      = ",f16.6," *")') exc
        write(*,'(" * Obliquity (deg)   = ",f16.6," *")') obl
        write(*,'(" * Obliquity (rad)   = ",f16.6," *")') oblr
        write(*,'(" * Long of perh(deg) = ",f16.6," *")') xlsper
        write(*,'(" * Long of perh(rad) = ",f16.6," *")') xlsperp
        write(*,'(" * Long at v.e.(rad) = ",f16.6," *")') lambm0
        write(*,'(" ****************************************")')
c      end if
      
     

      return
      end
